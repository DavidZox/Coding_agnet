import re
import os
from collections import OrderedDict, namedtuple

def get_data_path(path):
    """Construct a file path by joining the root directory with a given path."""
    return os.path.join(os.path.dirname(__file__), path)

            line = line.strip()

                continue
            
            parts = line.split(sep)
            if not parts:
                continue
                
            row_dict = {}

            # or we map based on index if usecols is provided.

                    if col_idx < len(parts):

def build_index():
    """Loads data from text files and creates an index."""
    
    # 1. Load Nationalities
    nationalities = {}
    nat_path = get_data_path('data_file/nationalities.txt')
    if os.path.exists(nat_path):
        with open(nat_path, 'r', encoding='utf-8') as f:
            for line in f:
                line = line.strip()
                if line and ':' in line:
                    nat, code = line.split(':')
                    nationalities[nat.lower()] = code

    # 2. Load Country Info
    countries = {}
    country_info_path = get_data_path('data_file/countryInfo.txt')
    if os.path.exists(country_info_path):
        with open(country_info_path, 'r', encoding='utf-8') as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                parts = line.split('\t')
                if len(parts) >= 5:
                    iso = parts[0]
                    iso3 = parts[1]
                    name = parts[4]
                    countries[iso] = name
                    countries[iso3] = name
                    countries[name] = iso # Map name back to ISO for filtering

    # 3. Load Cities
    cities = {}
    cities_path = get_data_path('data_file/cities15000.txt')
    if os.path.exists(cities_path):
        with open(cities_path, 'r', encoding='utf-8') as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                parts = line.split('\t')
                if len(parts) >= 14:
                    city_name = parts[1]
                    country_code = parts[13]
                    # Add city name and its aliases (from parts[3]) to the index
                    aliases = parts[3].split(',')
                    for alias in aliases:
                        alias = alias.strip()
                        if alias:
                            cities[alias.lower()] = (city_name, country_code)

    # 4. Load City Patches
    city_patches = {}
    patches_path = get_data_path('data_file/citypatches.txt')
    if os.path.exists(patches_path):
        with open(patches_path, 'r', encoding='utf-8') as f:
            for line in f:
                line = line.strip()
                if line and '\t' in line:
                    city, code = line.split('\t')
                    city_patches[city.lower()] = code

    Index = namedtuple('Index', ['nationalities', 'cities', 'countries', 'city_patches'])
    return Index(nationalities, cities, countries, city_patches)

class GeoText:
    def __init__(self, text, country=None):
        self.text = text
        self.country = country
        self.index = build_index()
        
        self.countries = []
        self.cities = []
        self.nationalities = []
        self.country_mentions = OrderedDict()
        
        # Pre-compile regex for finding words (candidates)
        self.city_regex = re.compile(r'\b[A-Z][a-z]+(?:\s[A-Z][a-z]+)*\b|\b[a-z]+\b')
        
        self._run_extraction()

    def _run_extraction(self):
        candidates = self._find_candidates(self.text)
        self.countries = self._extract_countries(candidates)
        self.nationalities = self._extract_nationalities(candidates)
        self.cities = self._extract_cities(candidates, self.country)
        self._calculate_country_mentions()

    def _find_candidates(self, text):
        # Find all words/sequences that could be names
        return re.findall(r'\b[\w\u00C0-\u017F]+\b', text)

    def _extract_countries(self, candidates):
        found_countries = []
        for cand in candidates:
            low_cand = cand.lower()
            # Check if candidate is a country name or ISO code in our index
            if cand in self.index.countries:
                found_countries.append(cand)
            elif low_cand in self.index.countries:
                # If it's a name, we want the original case or the name from index
                found_countries.append(self.index.countries[low_cand])
        return list(dict.fromkeys(found_countries))

    def _extract_cities(self, candidates, country_filter):
        found_cities = []
        
        # Resolve country filter to ISO code
        target_iso = None
        if country_filter:
            if country_filter in self.index.countries:
                target_iso = self.index.countries[country_filter]
            elif country_filter.upper() in self.index.countries:
                target_iso = country_filter.upper()

        for cand in candidates:
            low_cand = cand.lower()
            
            # Check patches first
            if low_cand in self.index.city_patches:
                if target_iso is None or self.index.city_patches[low_cand] == target_iso:
                    found_cities.append(cand)
                    continue

            # Check main cities index
            if low_cand in self.index.cities:
                city_name, country_code = self.index.cities[low_cand]
                if target_iso is None or country_code == target_iso:
                    found_cities.append(city_name)
                    
        return list(dict.fromkeys(found_cities))

    def _extract_nationalities(self, candidates):
        found_nationalities = []
        for cand in candidates:
            if cand.lower() in self.index.nationalities:
                found_nationalities.append(cand)
        return list(dict.fromkeys(found_nationalities))

    def _calculate_country_mentions(self):
        # Count how many times each country (by name) appears in the text
        # We use the text to find occurrences of country names
        words = re.findall(r'\b[\w\u00C0-\u017F]+\b', self.text)
        for word in words:
            low_word = word.lower()
            # Check if word is a country name or ISO code
            if word in self.index.countries:
                name = self.index.countries[word]
                self.country_mentions[name] = self.country_mentions.get(name, 0) + 1
            elif low_word in self.index.countries:
                name = self.index.countries[low_word]
                self.country_mentions[name] = self.country_mentions.get(name, 0) + 1
