import os
import re
from collections import OrderedDict, namedtuple

def get_data_path(path):
    """Construct a file path by joining the root directory with a given path."""
    return os.path.join(os.path.dirname(__file__), path)

def read_table(filename, usecols, sep='\t', comment='', encoding='utf-8', skip=0):
    """Parses data files to create dictionaries mapping terms to their corresponding values."""
    data = {}
    path = get_data_path(filename)
    if not os.path.exists(path):
        return data
    
    with open(path, 'r', encoding=encoding) as f:
        lines = f.readlines()[skip:]
        for line in lines:
            if not line.strip() or line.startswith(comment):
                continue
            parts = line.strip().split(sep)
            if len(parts) > 0:
                # Create a dictionary for the row based on usecols
                row_dict = {}
                for col_idx in usecols:
                    if col_idx < len(parts):
                        row_dict[col_idx] = parts[col_idx]
                
                # We need a way to map a key to the rest of the data. 
                # For simplicity in this implementation, we'll assume the first used col is the key.
                if usecols:
                    key_idx = usecols[0]
                    key = parts[key_idx]
                    data[key] = row_dict
    return data

def build_index():
    """Loads data from text files and creates an index."""
    # Load countries
    # countryInfo.txt: ISO(0), ISO3(1), ..., Country(4)
    country_info = read_table('data_file/countryInfo.txt', usecols=[0, 1, 4])
    countries = {v[0]: v[4] for k, v in country_info.items()} # ISO -> Name
    countries_by_name = {v[4]: k for k, v in country_info.items()} # Name -> ISO
    
    # Load cities
    # cities15000.txt: ID(0), Name(1), ..., CountryCode(10)
    # Note: The requirement says parts[10] is country code.
    cities_data = {}
    city_path = get_data_path('data_file/cities15000.txt')
    if os.path.exists(city_path):
        with open(city_path, 'r', encoding='utf-8') as f:
            for line in f:
                parts = line.strip().split('\t')
                if len(parts) >= 11:
                    city_name = parts[1]
                    country_code = parts[10]
                    if city_name not in cities_data:
                        cities_data[city_name] = []
                    cities_data[city_name].append(country_code)
    
    # Load nationalities
    # nationalities.txt: nationality:ISO
    nationalities = {}
    nat_path = get_data_path('data_file/nationalities.txt')
    if os.path.exists(nat_path):
        with open(nat_path, 'r', encoding='utf-8') as f:
            for line in f:
                if ':' in line:
                    nat, iso = line.strip().split(':')
                    nationalities[nat] = iso

    # Load city patches
    # citypatches.txt: city	ISO
    city_patches = {}
    patch_path = get_data_path('data_file/citypatches.txt')
    if os.path.exists(patch_path):
        with open(patch_path, 'r', encoding='utf-8') as f:
            for line in f:
                parts = line.strip().split('\t')
                if len(parts) == 2:
                    city_patches[parts[0]] = parts[1]

    Index = namedtuple('Index', ['countries', 'countries_by_name', 'cities', 'nationalities', 'city_patches'])
    return Index(countries, countries_by_name, cities_data, nationalities, city_patches)

class GeoText:
    def __init__(self, text, country=None):
        self.text = text
        self.country = country
        self.index = build_index()
        
        self.countries = []
        self.cities = []
        self.nationalities = []
        self.country_mentions = OrderedDict()
        
        # Regex to find words (candidates)
        self.city_regex = re.compile(r'\b[A-Z][a-z]+\b|\b[a-z]+\b')
        
        candidates = self._find_candidates(text)
        self.countries = self._extract_countries(candidates)
        self.cities = self._extract_cities(candidates, country)
        self.nationalities = self._extract_nationalities(candidates)
        self._calculate_country_mentions()

    def _find_candidates(self, text):
        return self.city_regex.findall(text)

    def _extract_countries(self, candidates):
        found_countries = []
        for cand in candidates:
            if cand in self.index.countries_by_name:
                found_countries.append(self.index.countries_by_name[cand])
        return list(dict.fromkeys(found_countries))

    def _extract_cities(self, candidates, country):
        found_cities = []
        for cand in candidates:
            # Check patches first
            target_country_code = self.index.city_patches.get(cand)
            
            if target_country_code:
                if country is None or target_country_code == country:
                    found_cities.append(cand)
            elif cand in self.index.cities:
                # Check if any of the city's country codes match
                city_country_codes = self.index.cities[cand]
                if country is None or country in city_country_codes:
                    found_cities.append(cand)
        return list(dict.fromkeys(found_cities))

    def _extract_nationalities(self, candidates):
        found_nationalities = []
        for cand in candidates:
            if cand in self.index.nationalities:
                found_nationalities.append(cand)
        return list(dict.fromkeys(found_nationalities))

    def _calculate_country_mentions(self):
        # Count how many times each country name appears in the text
        # Based on the requirement, we use the extracted countries list or scan text
        # We'll count occurrences of country names found in the text.
        for name in self.index.countries_by_name.keys():
            # Use regex to find whole word matches of the country name
            count = len(re.findall(r'\b' + re.escape(name) + r'\b', self.text))
            if count > 0:
                self.country_mentions[name] = count
